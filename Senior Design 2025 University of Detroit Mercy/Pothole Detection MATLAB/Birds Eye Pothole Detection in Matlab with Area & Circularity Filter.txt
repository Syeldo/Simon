clear;clear all;clc;


% ROS 2 Node and Topics Initialization
imageTopic = '/camera/image';%'/camera/image';
Node = ros2node('/lanenode',30);
imageSub = ros2subscriber(Node, imageTopic);
lanePub = ros2publisher(Node, "/camera/birdsEye", "sensor_msgs/Image");
% Predefined Constants
%sobel_x = [1, 0, -1; 2, 0, -2; 1, 0, -1]; % Sobel filter for x-gradient
topCropValue=250; % Adjustable crop offset
botCropValue=300; % Adjustable crop offset
sobel_x = [1, 0, -1; 2, 0, -2; 1, 0, -1]; % Sobel filter for x-gradient


while true
    % Receive Image Data
    [scanData, status, ~] = receive(imageSub, 10);

    if ~status
        disp('Failed to receive image.');
        continue;
    end

    msgOut = rosReadImage(scanData, "Encoding", 'bgr8');
    
    X=[280 970 1600 -300];
    %Y=[267.7637 265.2164 432.0622 430.7886];
    Y=[120 120 350 350];
    originalPoints = [X(1), Y(1); X(2), Y(2); X(3), Y(3); X(4), Y(4)];



    croppedImage=msgOut(topCropValue:end,:,:);
    %croppedImage=croppedImage(1:botCropValue,:,:);
    
    figure(1);imshow(croppedImage);hold on;
    plot(originalPoints(:,1), originalPoints(:,2), 'ro', 'MarkerSize', 10, 'LineWidth', 2);
    hold off;

    outputPoints = [1, 1; 880, 1; 880, 700; 1, 700];
    % Create the projective transformation matrix
    tform = fitgeotrans(originalPoints, outputPoints, 'projective');

    outputRef = imref2d([700,880]);
    % Apply the transformation
    correctedImage = imwarp(croppedImage, tform, 'OutputView',outputRef);

    % Display the result
    figure(2);imshow(correctedImage);title('outputref sized version');
    
    % Compute the inverse transformation which is useful for 
    % putting points back on the orginal image
    %inverseTform = invert(tform);
    
    % Transform the original points to the corrected image
    % this is redundant since these are mapped to the output point by the
    % transform
    transformedPoints = transformPointsForward(tform, originalPoints);


    hold on;
    plot(transformedPoints(:,1), transformedPoints(:,2), 'ro', 'MarkerSize', 10, 'LineWidth', 2);
    hold off;

Image_grey=rgb2gray(correctedImage);
%imshow(Image_grey);
ContrastedFrame=imadjust(Image_grey,[0.75, 1.0],[0, 1.0]);
%imshow(ContrastedFrame);
binary=imbinarize(ContrastedFrame);
%imshow(binary);

[BW_out,properties] = filterRegions(binary);

figure(1);imshow(BW_out);

%SE=strel('rectangle',[5 15]);
%openedFrame=imopen(ContrastedFrame,SE);
%imshow(openedFrame);
%openedFrame(openedFrame <= 150) = 0; % Thresholding
%imshow(openedFrame);
%GradientFrame=imfilter(openedFrame,sobel_x,'replicate','same','conv');
%imshow(GradientFrame);
%GradientFrame(GradientFrame <= 55) = 0; % Thresholding
%imshow(GradientFrame);



    %save("XformValues.mat","tform","inverseTform","originalPoints","outputPoints","outputRef");

    %figure(1);imshow(morph);title('publish');

    % Publish Processed Image
    %laneMsg = ros2message(lanePub);
    %laneMsg = rosWriteImage(laneMsg, rgbOut, "Encoding", "rgb8");
    %laneMsg.header.FrameId = 'camera';
    %send(lanePub, laneMsg);

end





function [BW_out,properties] = filterRegions(BW_in)

% Auto-generated by imageRegionAnalyzer app on 20-Mar-2025
%---------------------------------------------------------

BW_out = BW_in;

% Filter image based on image properties.
BW_out = bwpropfilt(BW_out,'Circularity',[0.7, 1]);

% Get properties.
properties = regionprops(BW_out, {'Area', 'Circularity', 'Eccentricity', 'EquivDiameter', 'EulerNumber', 'MajorAxisLength', 'MinorAxisLength', 'Orientation', 'Perimeter'});
end
